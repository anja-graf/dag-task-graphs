\subsection{Graphgenerierung mit Erdős–Rényi Modell}
\begin{frame} {\subsecname}
	\begin{center}
		\Large\textbf{Zwei Varianten des Modells:}
	\end{center}
	\begin{columns}
		\onslide<2->{
			\column{0.55\textwidth}
			$G(n,p)$ \\
			n = Anzahl an Knoten \\
			p = Wahrscheinlichkeit für Kanten
			\begin{itemize}
				\item Iteriere über geshuffelte Knotenvektoren, um Knoten i, j für jede mögliche Kante $e=\{i,j\} $ zu erhalten ($i \ne j$)
				\item Füge diese Kante e mit Wahrscheinlichkeit p hinzu
			\end{itemize}
		}
		\onslide<3->{

			\column{0.45\textwidth}
			$G(n,m)$ \\
			n = Anzahl an Knoten \\
			m = Anzahl an Kanten
			\begin{itemize}
				\item Wähle zwei zufällige Knoten aus
				\item Füge Kante zwischen Knoten hinzu, falls neue Kante
				\item Iteriere, bis Anzahl m erreicht
			\end{itemize}
		}
	\end{columns}

	\vspace*{1em}
	\onslide<4->{
		Insgesamt sind beide Varianten noch ungerichtet und prüfen nicht auf Kreisfreiheit!
	}
\end{frame}
\begin{frame} {\subsecname}
	\begin {columns}
	\begin {column}{0.6\textwidth}
	Daher:
	\begin{itemize}
		\item[1.]<2-> Füge gerichtete, statt ungerichtete Kanten ein
		\item[2.]<3-> Prüfe beim Hinzufügen ob eine der beiden Kanten $(i,j)$ oder $(j,i)$ bereits existiert
		\item[3.]<4-> Prüfe, ob es bereits Pfad $j $ nach $i$ gibt mittels rekursiven Tiefensuche
		\item<5-> Falls ja, würde neue Kante $e=(i,j) $ Kreis schließen
		      \onslide<6->{
			      \begin{itemize}
				      \item Darf nicht eingefügt werden
			      \end{itemize}
		      }
		      \onslide<7-> {

		      }
	\end{itemize}

\subsection{Uniprocessor Scheduling}
\begin{frame} {\subsecname}
	Ziel: Ausführungsreihenfolge finden
	\begin{itemize}
		\item Dabei sollen Abhängigkeiten berücksichtigt werden
		\item Also falls Kante $(i,j) \in E $ für Graph $G = (V,E)$ muss Task i vor Task j ausgeführt worden sein
		\item Dafür generieren wir zu Beginn eine Hash Map
		      \begin{itemize}
			      \item Keys: Knoten bzw. Task
			      \item Values: Jeweils eine Liste aller Vorgänger
			            (z.B.  $j \in$ Value Liste von Key i)
		      \end{itemize}
		\item Dann fügen wir iterativ bereitstehende Tasks mit der höchsten Priorität in den Schedule ein
		      \begin{itemize}
			      \item Task ist bereit/ready, falls voher keine abhängigen Tasks mehr ausgeführt werden müssen
		      \end{itemize}


	\end{itemize}
\end{frame}
\begin{frame} [fragile]{Pseudocode für \subsecname}
	\begin{center}
		\fbox{%
			\begin{minipage}{0.9\linewidth}
				\begin{algorithm}[H]
					\SetAlgoLined
					\footnotesize
					\onslide<1->{
						$schedule\gets []$,
						$ready\gets $ \{\},
						$pre\gets $ get\_predecessor\_hash\_map()\\
						\ForAll{$key, value \in pre : value =  \emptyset$}
						{
							$ready.add(key)$\
						}
						// throw error if ready = $\emptyset$ \\
					}
					\onslide<2->{
						\While{ready $\ne \emptyset$}{
							highest\_priority $\gets $ ready.get\_min() \\
							ready.remove(highest\_priority) \\
							pre.remove(highest\_priority) \\
							schedule.append(highest\_priority) \\
							\onslide<3->{
								\ForAll{$key, value \in pre : highest\_priority \in value $}
								{
									$value \gets value \setminus highest\_priority$ \\
									\If{$value = \emptyset$}{ready.add(key)}
								}
							}
						}
						\Return{schedule}
					}
				\end{algorithm}
			\end{minipage}}
	\end{center}
\end{frame}

%\subsection{Utilization mit Uunifast Algorithmus}
\begin{frame}{\subsecname}
	\begin{itemize}
		\item Weitere Anforderung an die Parameterverteilung: \\
		      Berücksichtigung der Auslastung von Tasks
		\item Utilization für Task i definiert als $U_i = \frac{C_i} { T_i }$
		      \begin{itemize}
			      \item dabei $T_i$ Periode, d.h. fester Zeitabstand, in welchem die Task i wiederholt ausgeführt werden soll
		      \end{itemize}
		\item Gesamtauslastung des Prozessors $U = \sum_{i = 1}^{n} U_i = \sum_{i = 1}^{n} \frac{C_i}{T_i}$
		      \begin{itemize}
			      \item \glqq Wie viel Prozessorzeit pro Zeiteinheit benötigen alle Tasks zusammen?\grqq
			      \item Wenn $U > 1 $: Prozessor ist überlastet und Tasks sind nicht schedulable/echtzeitfähig
		      \end{itemize}
		\item Im Programm kann zwischen Uunifast und Drs Algorithmus gewählt werden
		      \begin{itemize}
			      \item Dafür wird Parameter $U$ mitgegeben und daraus ein Vektor $U_1,U_2, ...,U_n$ berechnet
			      \item Die entsprechenden Werte werden für jede Task i angepasst, damit das Verhältnis mit $U_i$ übereinstimmt
		      \end{itemize}
	\end{itemize}
\end{frame}
\begin{frame} [fragile]{\subsecname ~mit \subsubsecname}
	\begin{lstlisting} [language=Rust, style=colouredRust,xleftmargin=15pt, frame=single]
fn uunifast(&self, u: f64) -> Vec<f64> {
	assert!(u <= 1.0 && u >= 0.0, 
	  "The processor utilization should be value between 0 and 1");
	let mut sum_u = u;
	let mut rng = rand::thread_rng();
	let mut vect_u: Vec<f64> = Vec::new();
	for i in 0..self.nodes.len() - 1 {
		let next_sum_u = sum_u * rng.r#gen::<f64>()
			.powf(1.0 / (self.nodes.len() - i) as f64);
		vect_u.push(sum_u - next_sum_u);
		sum_u = next_sum_u;
	}
	vect_u.push(sum_u);
	assert!(u - vect_u.iter().sum::<f64>().abs() < 1e-6);
	vect_u
}\end{lstlisting}

\begin{frame} {\subsecname ~mit \subsubsecname}
	Constraints für generierte Graphen  \\
	Gilt $\forall i \in \{0..n-1\}$, n = Anzahl der Tasks, i = Index der Task im Schedule
	\begin{itemize}
		\item $C_i  \in \{10..100\} $ \\
		      Die Computation time muss für alle Knoten zwischen 10 und 100 liegen.
		\item $d_i \ge C_i + s_i $ \\
		      Die Deadline muss spät genug sein, damit die Task vorher ausgeführt werden kann.
		\item $s_i \ge d_{i-1}, $ wobei $s_0 = 0$ \\
		      Eine Task, welche früher gescheduled ist, muss beendet worden sein bevor die nächste startet. Zur Vereinfachung wird $s_i = d_{i-1}$ gesetzt.
		\item $U_i =  \frac{C_i}{T_i}$ \\
		      Das Verhältnis aus Computation time und Periode muss der berechneten Utilization entsprechen. Dabei wird der Einfachheit halber $T_i = d_i$ gleichgesetzt.
	\end{itemize}
\end{frame}
